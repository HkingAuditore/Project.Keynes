模型速览

demand = 家庭需求EMA + 商业需求EMA
flow = demand + 供给EMA
excess = clamp((demand − supply)/max(1000,flow), −1, +1)
inventory = clamp((目标库存 − stock)/max(1000,目标库存), −1, +1)  （仅 storage_mode==0）
shortage = clamp(短缺率, 0, 1)
cost = clamp((锚−价)/max(锚,价), −1, +1) × min(1, 供给/demand)  （仅非货币品且有锚；锚>价且缺货→设硬底价）
idle = clamp((默认价−价)/max(1,max(默认价,价)), −1, +1)  （仅 demand==supply==stock==0）
total = excess·w_excess + inventory·w_inv + shortage·w_short + cost·w_cost + idle·w_idle
elasticity = clamp(弹性, 0.25, 4)
change_q16 = (total / elasticity) × 价格调节率
change' = clamp(change_q16, −日最大跌, +日最大涨)
next = 价 + 价 × (change' × epoch_days) / 65536
final = clamp(next, min_price, max_price)
