[i] = cloud;
        if (OUT_CW != nullptr) {
            OUT_CW[i] = cloud_water;
        }
        OUT_PRE[i] = precip;
        OUT_INS[i] = instability;
        OUT_INT[i] = intensity;
        OUT_CNV[i] = convergence;
        if (use_next_outputs) {
            OUT_TYP_I32[i] = int32_t(wt);
        } else {
            OUT_TYP[i] = display_wt;
            OUT_FIN[i] = 1; // weather_field_initialized = true
        }
    }
}
